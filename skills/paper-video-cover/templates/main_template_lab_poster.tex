% 〈TITLE1UNI〉= 机构眉题（如 纽约大学，宽度预算 40 单位，CJK=2/其他=1，可留空则删行）；
% 〈TITLE1UNIEN〉= 机构英文小字（如 NEW YORK UNIVERSITY，≤40 字符，可留空则删行）；
% 〈TITLE1UNICOLOR〉= 机构 logo 品牌色（#RRGGBB，缺省 000000 黑，见 definecolor 行）；
% 〈TITLE1PI〉= 教授姓名（如 Prof. Yann LeCun，≤30 字符，必填）；
% 〈TITLE2〉〈TITLE3〉宽度预算各 30 单位（CJK=2/其他=1），超限由生成器拒绝。

\documentclass[aspectratio=1610,10pt]{ctexbeamer}
\usetheme{sustech}

\usepackage{booktabs}
\usepackage{geometry}

\usepackage{tikz}
\usetikzlibrary{shadows}
\graphicspath{{figures/}}

% --- 全幅背景（与 deck 可视化分析分节页一致；无 background.png 时回退浅底）-----
\setbeamertemplate{background canvas}{%
  \IfFileExists{figures/background.png}{%
    \includegraphics[width=\paperwidth,height=\paperheight]{figures/background.png}}{%
    \color{white!90!black}\vrule width\paperwidth height\paperheight}}

% --- 网格常量（4:3 容器浮于 16:10 页面；方案 B：容器 1620×1215 @ 2324×1440）-------
% 容器：宽 0.6971\textwidth（≈22.3cm）× 高 0.8333\textheight，居中
\newcommand{\containerwidth}{0.6971\textwidth}
% 统一间距（G=60px：横 60/2324≈2.58%，竖 60/1440≈4.17%）
\newcommand{\postergap}{0.0258\textwidth}
\newcommand{\postervgap}{0.0417\textheight}
% 等分块宽（字面量；节点总宽 = text width + 2×inner sep(12pt≈0.0265\textwidth)）：
\newcommand{\boxpad}{0.0265\textwidth}
\newcommand{\boxwfull}{0.6706\textwidth}      % 整行：外缘 = 0.6971
\newcommand{\blockwtwo}{0.3091\textwidth}     % 两块行：外缘各 = 0.3356（(0.6971-0.0258)/2）
% 行高（px/1440）：180 / 260 / 310
\newcommand{\rowhone}{0.1250\textheight}
\newcommand{\rowhtwo}{0.1806\textheight}
\newcommand{\rowhthree}{0.2153\textheight}

% --- 阴影圆角块 ---------------------------------------------------------------
% #1 可选宽度（默认容器宽）；#2 最小高度；#3 内容
\newcommand{\posterbox}[3][\containerwidth]{%
  \begin{tikzpicture}%
    \node[fill=white!85!black, rounded corners=10pt, inner sep=12pt,
          text width=#1, align=center, minimum height=#2,
          drop shadow={shadow xshift=6pt, shadow yshift=-6pt, opacity=0.5, fill=black}]
          {#3};%
  \end{tikzpicture}}

% --- 眉题品牌色（机构 logo 色；无 logo 用黑 #000000）-------------------------
% 该行默认 000000 以保证模板可裸编译；生成器见 〈TITLE1UNICOLOR〉 时把
% 前后两个 000000 一起换成机构品牌色（#RRGGBB）。
\definecolor{instbrand}{HTML}{000000}% 〈TITLE1UNICOLOR〉
\definecolor{nyugrey}{HTML}{8A8F98}      % 英文小字灰
% 行内并排：\posterrow{块A}{块B}{块C}（1–3 个参数，统一间距）
% 用整行 \hbox to \textwidth + \hss 两端胶水居中：\hbox to 容器宽 不是段落行，
% 不会被 \centering 居中（会贴左边缘）；整行 hbox 才保证内容居中。
\newcommand{\posterrow}[3]{%
  \hbox to \textwidth{\hss
    #1\ifx\\#2\\\else\hspace{\postergap}#2\fi
    \ifx\\#3\\\else\hspace{\postergap}#3\fi
  \hss}}
